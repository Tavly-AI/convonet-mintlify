import { useState } from 'react';

# Getting Started with MDX

MDX allows you to write JSX directly inside your Markdown documents. This enables interactive elements like stateful React components alongside standard content.

---

## 1. Interactive Component Example

Below is an example of an inline interactive React component rendered directly in the document:

export const Counter = () => {
  const [count, setCount] = useState(0);
  return (
    <div style={{
      padding: '1rem',
      borderRadius: '8px',
      border: '1px solid #333',
      background: '#121212',
      color: '#fff',
      margin: '1rem 0',
      maxWidth: '300px'
    }}>
      <p style={{ margin: '0 0 0.5rem 0', fontWeight: 'bold' }}>Current Count: {count}</p>
      <button 
        onClick={() => setCount(count + 1)}
        style={{
          padding: '0.5rem 1rem',
          borderRadius: '4px',
          border: 'none',
          background: '#e5b800',
          color: '#000',
          fontWeight: 'bold',
          cursor: 'pointer'
        }}
      >
        Increment
      </button>
    </div>
  );
};

<Counter />

export function calculateReadTime(text: string): number {
  const wordsPerMinute = 200;
  const words = text.split(/\s+/).length;
  return Math.ceil(words / wordsPerMinute);
